class Solution {
    public int[] leftRightDifference(int[] nums) {
        int n = nums.length;

        int[] left = new int[n];
        int[] right = new int[n];
        int[] answer = new int[n];

        left[0] = 0;
        right[n - 1] = 0;

        for (int i = 1; i < n; i++) {
            left[i] = nums[i - 1] + left[i - 1];
        }

        for (int j = n - 2; j >= 0; j--) {
            right[j] = nums[j + 1] + right[j + 1];
        }
        for (int k = 0; k < n; k++) {
            answer[k] = Math.abs(right[k] - left[k]);
        }

        return answer;
    }
}
